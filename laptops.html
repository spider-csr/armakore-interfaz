const PRECIO_BASE = 1899; function seleccionarOpcion(elemento, categoria, precio) { const padre = elemento.parentElement; padre.querySelectorAll('.config-option').forEach(opcion => { opcion.classList.remove('option-selected'); }); elemento.classList.add('option-selected'); // Guarda SOLO el precio MODIFICADOR, no el valor absoluto preciosSeleccionados[categoria] = precio; // Recalcula desde cero actualizarPrecioTotal(); } function actualizarPrecioTotal() { let total = PRECIO_BASE; Object.values(preciosSeleccionados).forEach(modificador => { total += modificador; }); const totalPriceEl = document.getElementById('total-price'); totalPriceEl.textContent = '$' + total.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 }); // Opcional: Animación visual cuando cambia el precio totalPriceEl.style.transform = 'scale(1.1)'; setTimeout(() => totalPriceEl.style.transform = 'scale(1)', 200); } // Inicializar con valores por defecto actualizarPrecioTotal();
